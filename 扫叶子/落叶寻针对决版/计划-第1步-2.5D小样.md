# 落叶寻针 v0.12 第 1 步：2.5D 小样 · 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 做出可试玩的斜俯视 2.5D 叶堆小样：捡叶、深度明暗、露出比例、按住拔针、反光，能在本机 OBS 浏览器源里透明显示并通过「交互」窗口操作，交给用户试镜头角度和手感。

**Tech Stack:** 原生 JavaScript（只用 Chromium 127 支持的特性）、WebGL2、Node 26 自带 `node:test`、无第三方依赖。

**设计依据:** `设计.md`（同目录）。本计划只覆盖设计第 12 节的第 1 步；第 2–4 步在用户试完小样、镜头和手感定下来之后，再各写一份计划（见文末「后续阶段」）。

(function(root,factory){if(typeof module==='object'&&module.exports)module.exports=factory();else root.LeafCamera=factory();})(typeof globalThis!=='undefined'?globalThis:this,function(){
'use strict';
const STAGE_W=1440,STAGE_H=810;
const dot=(a,b)=>a[0]*b[0]+a[1]*b[1]+a[2]*b[2];
class Camera{
  constructor(opts={}){this.set(opts);}
  set({elevation=this.elevation??60,focal=this.focal??1500,distance=this.distance??1500,center=this.center??[720,500],target=this.target??[0,0,0]}={}){
    this.elevation=Math.min(75,Math.max(45,Number(elevation)||60));
    this.focal=focal;this.distance=distance;this.center=center.slice();this.target=target.slice();
    const t=this.elevation*Math.PI/180,c=Math.cos(t),s=Math.sin(t);
    this.forward=[0,c,-s];this.right=[1,0,0];this.up=[0,s,c];
    this.position=[this.target[0],this.target[1]-distance*c,this.target[2]+distance*s];
    return this;
  }
  toCamera(p){const d=[p[0]-this.position[0],p[1]-this.position[1],p[2]-this.position[2]];return[dot(d,this.right),dot(d,this.up),dot(d,this.forward)];}
  project(p){const [x,y,z]=this.toCamera(p);return{x:this.center[0]+this.focal*x/z,y:this.center[1]-this.focal*y/z,depth:z,scale:this.focal/z};}
  ray(sx,sy){
    const dx=(sx-this.center[0])/this.focal,dy=-(sy-this.center[1])/this.focal,R=this.right,U=this.up,F=this.forward;
    const d=[R[0]*dx+U[0]*dy+F[0],R[1]*dx+U[1]*dy+F[1],R[2]*dx+U[2]*dy+F[2]],n=Math.hypot(d[0],d[1],d[2]);
    return{origin:this.position.slice(),dir:[d[0]/n,d[1]/n,d[2]/n]};
  }
  viewProj(near=10,far=20000){
    const f=this.focal,[cx,cy]=this.center,C=this.position,row=v=>[v[0],v[1],v[2],-dot(v,C)];
    const Rr=row(this.right),Ur=row(this.up),Fr=row(this.forward),a=(far+near)/(far-near),b=-2*far*near/(far-near);
    const rows=[Rr.map((v,i)=>2*f/STAGE_W*v+(2*cx/STAGE_W-1)*Fr[i]),Ur.map((v,i)=>2*f/STAGE_H*v+(1-2*cy/STAGE_H)*Fr[i]),Fr.map((v,i)=>a*v+(i===3?b:0)),Fr];
    const m=new Float32Array(16);for(let r=0;r<4;r++)for(let c=0;c<4;c++)m[c*4+r]=rows[r][c];return m;
  }
}
function rayAtHeight(ray,z){const t=(z-ray.origin[2])/ray.dir[2];return{t,x:ray.origin[0]+ray.dir[0]*t,y:ray.origin[1]+ray.dir[1]*t,z};}
return{Camera,rayAtHeight,dot,STAGE_W,STAGE_H};
});
```

(function(root,factory){if(typeof module==='object'&&module.exports)module.exports=factory();else root.LeafLayout=factory();})(typeof globalThis!=='undefined'?globalThis:this,function(){
'use strict';
const W=1440,H=810,ALPHA_MIN=32;
function fit(w,h){const scale=Math.min(w/W,h/H);return{scale,x:(w-W*scale)/2,y:(h-H*scale)/2};}
function point(x,y,rect){return{x:(x-rect.left)*W/rect.width,y:(y-rect.top)*H/rect.height};}
function spriteComponents(data,w,h,alphaMin=ALPHA_MIN){
  const seen=new Uint8Array(w*h),queue=new Int32Array(w*h),best=new Array(9).fill(null);
  for(let p=0;p<w*h;p++){
    if(seen[p]||data[p*4+3]<alphaMin)continue;
    let read=0,end=1,minX=w,minY=h,maxX=0,maxY=0;queue[0]=p;seen[p]=1;
    while(read<end){
      const q=queue[read++],x=q%w,y=(q/w)|0;
      if(x<minX)minX=x;if(x>maxX)maxX=x;if(y<minY)minY=y;if(y>maxY)maxY=y;
      const nb=[x>0?q-1:-1,x<w-1?q+1:-1,y>0?q-w:-1,y<h-1?q+w:-1];
      for(const n of nb)if(n>=0&&!seen[n]&&data[n*4+3]>=alphaMin){seen[n]=1;queue[end++]=n;}
    }
    if(end<=20)continue;
    const col=Math.min(2,Math.floor((minX+maxX)/2/w*3)),row=Math.min(2,Math.floor((minY+maxY)/2/h*3)),k=row*3+col;
    if(best[k]&&best[k].area>=end)continue;
    const rw=maxX-minX+1,rh=maxY-minY+1,mask=new Uint8Array(rw*rh);
    for(let i=0;i<end;i++){const q=queue[i];mask[(((q/w)|0)-minY)*rw+(q%w-minX)]=1;}
    best[k]={x:minX,y:minY,w:rw,h:rh,area:end,mask};
  }
  return best;
}
return{width:W,height:H,ALPHA_MIN,fit,point,spriteComponents};
});
```

### Task 3: 叶堆基础 `pile.js`（生成、堆叠、高度网格）

**Files:**
- Create: `src/pile.js`
- Create: `tests/helpers.cjs`
- Test: `tests/pile-base.test.cjs`

数据约定：
- 叶片 `{id,x,y,z,yaw,cos,sin,size,type,shade,depth}`，都是水平薄片，`z` 为所在高度。叶片局部坐标 `u` 沿 `(cos,sin)`，`v` 沿 `(-sin,cos)`，取值 -0.5…0.5；掩码第 0 行是图片顶部，对应 `v=+0.5`（渲染时 `v` 轴正向也对应图片顶部，两边一致）。depth 是开局时被原始表面盖住的深度（最上层为 0），用于明暗。
- 叶子是软的，会贴着下面的表面：新叶高度 = 已经盖住它中心点的叶子数加 1，再乘以每层厚度 `thickness`。这样堆高和该处的层数成正比，不会因为「搁在最高点」而越堆越高。
- 叶子按穹顶分布撒在椭圆里：面积密度 ∝ 1−r²（中心最密，边缘降到 0），所以中心厚度约为平均的两倍，叶堆是一座圆丘而不是尖锥。
- 前后顺序统一用 `(z, id)`：高度大的在上，同高时后放的（id 大）在上。点选、露出比例和绘制都按这个顺序，所以「看到的就是点到的」。镜头在所有叶片上方，同一条视线总是先碰到更高的水平面。
- 空间哈希格 64 单位，叶片按外接圆半径 `0.71×size+8` 登记进所有覆盖到的格子；高度网格 8 单位，记录每格中心点上最高叶片的高度。

### Task 9: 小样装配 `app.js`

**Files:**

要点：
- **素材切片**：从图集取 6 片叶子和木桶，每个连通块只保留自身像素，缩放后放进 256×256 的格子（4×2 共 8 格）。叶片掩码直接从格子的透明度取 64×64（≥50% 算不透明），和渲染用同一张图，保证「看到的就是点到的」。
- **每帧**：拔针计时 → 反光（按间隔冒星点）→ 动画清理 → 绘制（针下方的叶子 → 埋着的针 → 针上方的叶子 → 木桶和飞行的叶子 → 拔出中或调试透视的针 → 星点）→ 针旁标签和统计。
- **按下**：先判断是否点在露出的针上（够阈值就开始拔，不够就提示「还拔不动」）；否则捡视线上最前面的叶子，0.3 秒内不能再捡。
- `window.__leafDebug` 供浏览器验收读取针的屏幕位置、读取像素、调用调试露出。
- 本步不做音效（设计第 12 节把音效放在第 4 步）。
